# Unions
is_special_personality = {
	OR = {
		has_ai_personality = peaceful_counterculture
		has_ai_personality = mirthful_bandits
		has_ai_personality = cunning_raiders
	}
}

is_megaunion = {
	has_authority = auth_corporate
	is_socialism = yes
}

is_socialist_government = {
	OR = {
		has_government = gov_trade_union_corporate
		has_government = gov_communist_dictatorship
		has_government = gov_anarchy_communism
		has_government = gov_popular_front
		has_government = gov_worker_coop
	}
}

is_fascist = {
	OR = {
		has_government = gov_national_socialism
		has_government = gov_jingoism
		AND = {
			has_ethic = ethic_fanatic_authoritarian
			is_capitalism = yes
			OR = {
				is_militarist = yes
				is_xenophobe = yes
			}
		}
	}
}

is_anti_fascist = {
	OR = {
		has_government = gov_anarchy_state
		has_government = gov_anarchy_communism
		has_government = gov_popular_front
		AND = {
			has_ethic = ethic_fanatic_egalitarian
			is_socialism = yes
			OR = {
				is_militarist = yes
				is_xenophile = yes
			}
		}
	}
}
